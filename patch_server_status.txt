  };

    const { error: prodError } = await supabase.from('produtos').select('id').limit(1);

    statusPayload.tables.banners = !banError;

      if (banError) errMsgs.push(`Tabela 'banners': ${banError.message}`);
      statusPayload.error = errMsgs.join(' | ');
